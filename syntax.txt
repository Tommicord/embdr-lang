# Embedrite syntax

## 1. Hello world

use <embdr/std>;

// "#addr(main)" places the function at 0x00 — the implicit main entry
// point (only one can be defined). #addr(main) is the same as #addr(0x00).
#addr(main) function ::main() {
    // "std" is a namespace; "<name>" after "::" marks a macro call
    std::<println>("Hello, World!");
    ret 0;
}

## 2. Variables, Functions

function ::my_function() {
    int32 a = 10;
    int32 b = 10;
    let c = a + b;
    ret c;
}

## 3. Data types

function ::my_function() {
    int128 big_int = 1000000000000000000000int128;
    int512 very_big_int = big_int * 1024; // Uses AVX-512 registers if available, Maps to bitwise operation if missing multiply operator
    float64 my_float = 1.2345float64;
    float16 half_precision = 1.2float16;
    u8string utf8_string = u8"Hello, World!";
    u16string utf16_encoded = utf8_string.encode_utf16();
    bool my_bool = true;
    bool false_bool = false;
    std::<println>(utf16_encoded); // Hello, World!
}

## 4. Macros

// '$' starts a pattern and '^' ends it. String literals inside the
// pattern match literally; '$name' captures the matched text
// (identifiers/expressions, no data type). [<...>] marks where an identifier/expression goes.

macro ::<hello_world> {
        ($ "[<" $str ">]" ^) {
                std::<println>($str);
        }
}

## 5. Structs

// 1. Default aligned
struct ::my_container {
        pub {
                a: bool;
                b: int32;
                c: [?]u8string; // Unknown string size
                // Constructors/Destructors are implicitly generated by the compiler
        }
}

// 2. Explicit alignment
#alignas(16) struct ::my_container {
        pub {
                a: bool;
                b: int32;
                c: []u8string;
                d: [10]u8string;

                #constructor self() {
                        self.a = true;
                        self.b = 10;
                        self.c = "Hello";
                        self.d = "abcdefghij"; // Must be exactly 10 chars
                }
                #destructor() default;
        }
}

## 7. Embeddings

// The embeddings are syntactic sugar for doing a task when a function starts and ends
static std::lazy_locker<std::once<uint32>> my_var = std::once(
        static function(addr) {
                *addr = 1000;
        }
);

embed ::do_print {
        pub {
                fn do_stuff() {
                        uint32 ref val = my_var.lock(); // Calls the lazy locker closure
                        std::<println>(val);
                        val += 10;
                }
        }
        #constructor self() {
                self.do_stuff();
        }
        #destructor() default;
}

#do_print
function ::do_something() {
    // Do stuff.
}

#addr(main) function ::main() {
    ::do_something(); // my_var = 1000, then prints
    ::do_something(); // my_var = 1010, then prints
    ::do_something(); // my_var = 1020, then prints
    ret 0;
}

## 8. Operators and expressions

// Precedence, from tightest to loosest:
//
//   1  postfix      a.b   a(args)  a[args]  a?        (left)
//   2  unary        *p    &x     -x   !x   ~x         (right)
//   3  cast         x as T                              (left)
//   4  multiplicative  *  /  %                          (left)
//   5  additive       +  -                              (left)
//   6  shift          <<  >>                            (left)
//   7  bit and        &                                 (left)
//   8  bit xor        ^                                 (left)
//   9  bit or         |                                 (left)
//  10  range          ..  ..=                           (non-assoc)
//  11  comparison     ==  !=  <  <=  >  >=   is         (non-assoc)
//  12  logical and    &&
//  13  logical or     ||
//  14  default        ??          (maybe<T> extraction with fallback)
//  15  pipeline       |>          (f |> g(x) == g(f, x))
//  16  assignment     =  +=  -=  *=  /=  %=  &=  |=  ^=  <<=  >>=  (right)

// Arithmetic: integer + - * / % operate in the operand's type.
// Overflow is NEVER undefined: it wraps (two's complement) by default;
// use std::math::checked_add / checked_mul when trapping is wanted.
// Division by zero calls std::debug::panic (defined behavior, no UB).

function ::ops() -> int32 {
        int32 a = 10;
        a += 2;              // compound assignment
        a <<= 1;             // shift compound
        bool ok = a > 4 && a < 100;
        int32 low = a & 0xFF;  // bit and
        int32 sw  = a ^ 0x0F;  // bit xor
        float64 r = (a as float64) / 3.0;   // "as" converts between types
        maybe<uint32> m = ::find(a);
        uint32 v = m ?? 42;     // extract maybe or fallback (no null anywhere)
        ret a;
}

// String and char literals:
//   "hi"  u8"hi"   -> u8string        u16"hi" -> u16string
//   r"raw\ntext"    -> raw, no escapes
//   'x' '\n' '\x41' '\u{1F600}'       -> char (Unicode scalar, 32-bit)
// Numbers accept _ separators and a type suffix (the suffix IS the type name):
//   1_000_000  0xFF  0b1010  0o777  100int128  3.14float64  1.2float16  6.02e23

## 9. Control flow

function ::control(n: uint32) -> uint32 {
        // if: statement and expression (both forms are valid)
        if n == 0 {
                ret 0;
        }
        uint32 grade = if n > 100 { 100 } else { n };   // if-expression

        // while
        uint32 i = 0;
        while i < n {
                i += 1;
                if i == 13 {
                        continue;      // skip this iteration
                }
                if i > 10 {
                        break;         // exit the loop
                }
        }

        // loop = infinite, can yield a value through break
        uint32 ticks = 0;
        loop {
                if hw::timer_ready() {
                        break ticks;
                }
                ticks += 1;
        }

        // for over ranges (.. = half-open, ..= = inclusive)
        uint32 sum = 0;
        for k in 0 .. n {
                sum += k;
        }
        for k in 0 ..= 4 {
                sum += k;
        }

        // for over slices/arrays/vecs, with index sugar
        [4]uint32 data = [1, 2, 3, 4];       // array literal
        for x in data {
                sum += x;
        }
        for idx, x in data {               // idx, value
                sum += x + idx;
        }
        for x in data[1..] {               // sliced iteration
                sum += x;
        }

        // labels for outer break/continue
        :outer {
                for a in 0 .. 10 {
                        for b in 0 .. 10 {
                                if a * b > 20 {
                                        break :outer;
                                }
                        }
                }
        }
        // defer: runs when the scope exits (early ret included)
        let f = std::io::open("/log.txt", "w");
        defer std::io::close(f);
        std::io::writeln(f, "boot");
        ret sum;                           // close runs here
}

// match: expression or statement; arms are "pattern { expr }".
// Patterns: enum variants, literals, ranges, | alternatives,
//           bindings (name), guards ("if cond"), wildcard "_",
//           p"..." compile-time patterns (see §19).
enum ::color {
        pub {
                red, green, blue
        }
}

function ::describe(c: color) -> u8string {
        ret match c {
                red   { "Red" }
                green { "Green" }
                blue  { "Blue" }
                      { "Unknown" }     // default arm
        };
}

function ::range_class(x: uint32) -> u8string {
        ret match x {
                0        { "zero" }
                1 .. 10  { "small" }
                10 | 20  { "round" }    // alternatives
                _        { "big" }
        };
}

## 10. Error handling — Result and '?'

// There are no exceptions and no unwinding. Every fallible function
// returns result<T, E>; the '?' operator propagates Err to the caller.

use <embdr/std>;

function ::read_temp(bus: i2c::bus) -> result<float32, i2c::error> {
    let raw = bus.read_reg(0x00)?;     // on Err: ret Err(raw) immediately
    let calibrated = calibrate(raw)?;  // chains are free (monadic bind)
    ret Ok(calibrated as float32);
}

// maybe<T> replaces null entirely: some(v) | none
function ::find(key: uint32) -> maybe<uint32> {
        if key == 0 {
                ret none;
        } else {
                ret some(key * 10);
        }
}

## 11. Compile-time evaluation (cexpr)

// 1. IMPLICIT: any pure expression with constant operands is folded.
//    No keyword needed — this is a language guarantee, not a heuristic.
// 2. FORCED: `cexpr` on a binding demands evaluation; the compiler
//    errors if the value is not computable at compile time.

function ::comptime() {
        cexpr int32 a = 10 * 64;              // folded to 640
        cexpr mask = (1 << 16) - 1;           // folded to 0xFFFF
        cexpr f    = std::math::sqrt(2.0);    // std math is pure -> folded
        int32 implicit = 2 + 3 * a;           // also folded (no annotation)

        // Compile-time pointers: &global folds to a fixed absolute address
        static [256]uint8 buffer;
        cexpr base = &buffer;               // becomes a link-time constant
        cexpr tail = &buffer[255];          // base + 255, folded

        // cexpr for = unrolled at compile time (drives #unroll automatically):
        // each iteration's body is emitted once; pure bodies fold entirely.
        cexpr for i in 0 .. 4 {
                std::<println>(i);              // prints 0 1 2 3, zero loop code
        }

        // Compile-time reflection results are constants too (see §12)
        cexpr sz = ::<sizeof>(color);       // 1 (folded)
        ret;
}

## 12. Compile-time reflection

// Everything below is evaluated at compile time by built-in macros in the
// global scope (::name). Results are cexpr values: strings are literals,
// counts are numbers, metadata never reaches the binary unless you ask
// for it with #reflect (size first!).

// -- type-level builtins ------------------------------------------------
//   ::<type_of>(expr)      -> type        typeof an expression
//   ::<type_name>(T)       -> u8string    "my_container"
//   ::<sizeof>(T)          -> uint64      size in bytes
//   ::<alignof>(T)         -> uint64      required alignment
//   ::<is_same>(A, B)      -> bool
//   ::<is_struct>(T)       -> bool        (also is_enum, is_array, is_ref)
//   ::<is_signed>(T)       -> bool

// -- aggregate reflection ----------------------------------------------
//   ::<field_count>(T)     -> uint64
//   ::<field_at>(T, i)     -> field_info { name, type, offset, index }
//   ::<field_by_name>(T, "x") -> field_info  (compile error if missing)
//   ::<offset_of>(T, "x")  -> uint64

// -- enum reflection ----------------------------------------------------
//   ::<enum_len>(T)      -> uint64
//   ::<enum_at>(T, i)      -> value       (variant at index i)
//   ::<enum_name>(T, v)    -> u8string    "blue"
//   ::<enum_from_name>(T, "blue") -> value

function ::introspect() {
        cexpr for i in 0 .. ::<field_count>(my_container) {
                // Unrolled at compile time; each iteration prints a constant
                cexpr f = ::<field_at>(my_container, i);
                std::<println>("{} at offset {}", f.name, f.offset);
        }

        cexpr for i in 0 .. ::<enum_len>(color) {
                std::<println>(::<enum_name>(color, ::<enum_at>(color, i)));
        }
        ret;
}

// Works with generics too — reflection over an instantiation:
function ::dump<T>() {
        std::<println>("{} ({} bytes)", ::<type_name>(T), ::<sizeof>(T));
        cexpr for i in 0 .. ::<field_count>(T) {
                std::<println>("  .{} : {}", ::<field_at>(T, i).name,
                                           ::<type_name>(::<field_at>(T, i).type));
        }
        ret;
}

// Runtime reflection is OPT-IN so binaries stay small:
#reflect struct ::point {
        pub {
                x: float32;
                y: float32;
        }
}

// With #reflect the metadata is emitted into .rodata and usable at runtime:
//   std::reflect::type_name(point)       -> "point"
//   std::reflect::field_names(point)     -> ["x", "y"]
//   std::reflect::field_offset(point, 1) -> 4
// Without #reflect these calls are compile errors — metadata is never
// emitted "just in case". Doc comments (///) are kept only under #reflect.

## 13. Pointers, references and slices

//   T ref      non-null reference, always points to a valid, initialized
//              object (compiler-guaranteed, ROS-checked) — the default
//              way to share data
//   *T         raw pointer, only dereferenceable inside volatile { }
//   &x         address-of (folds to a constant when x is static/cexpr)
//   *p         dereference
//   []T        slice = pointer + length (bounds-checked by default)
//   [N]T       fixed-size array
//   [?]T       extern/unknown-size array (only at FFI boundaries)

function ::pointers(static_arr: [4]int32) -> int32 {
        int32 ref r = static_arr[0];     // reference into the array
        r = 42;                        // writes through, no null possible

        *int32 p = &r;                   // raw pointer (address known at compile time)
        volatile {
                int32 v = *p;                // deref of raw pointer requires volatile
        }

        []int32 slice = static_arr[..];  // slice of the whole array
        int32 total = 0;
        for x in slice {
                total += x;                // bounds are proven or checked — no UB
        }
        ret total;
}

## 14. volatile and portable asm

// Safe code cannot cause undefined behavior, period. `volatile` is the
// single escape hatch: it tells the compiler not to optimize, reorder or
// rewrite the marked block/object, and it is the only place where UB can
// happen. It is explicit, greppable, and takes three forms:

// 1. volatile { } — required to dereference raw pointers, call extern
//    functions, or perform inline asm. The safety invariants are the
//    programmer's responsibility inside; violating them is the ONLY
//    source of UB.
function ::poke(addr: uint64, val: uint32) {
        volatile {
                *uint32 p = addr as *uint32;
                *p = val;
        }
        ret;
}

// 2. volatile object — for MMIO. Never elided, never reordered, never cached.
volatile static uint32 uptime_ms;        // whole object is volatile (compiler will not vectorize it, UB allowed)
volatile {
        std::mem::write_volatile(reg as *uint32, 1uint32);  // explicit volatile store
        uint32 s = std::mem::read_volatile(reg as *uint32); // explicit volatile load
}

// 3. volatile(asm) — portable asm: a small, target-checked instruction
//    vocabulary. The same instruction compiles on every target that
//    implements it; otherwise it is a compile error (no silently-wrong
//    assembly).
volatile(asm) {
        nop           // no-op on every target
        dmb           // data memory barrier (validated)
        irqd          // spelled portably, lowered per target
}
// Target-specific escape hatch (non-portable by definition):
#target("thumbv7m")
volatile(asm) {
        cpsid i
}

## 15. Lambdas and closures

// Function type syntax: fn(params) -> ret
// Closure literals:     function(params) { ... ret x; }
// Captures are by-value; shared mutable state goes through `ref`.

function ::lambdas() {
        fn(int32, int32) -> int32 add = function(a, b) { ret a + b; };
        let doubled = std::iter::map([1, 2, 3], function(x) { ret x * 2; });

        int32 counter = 0;
        let tick = function() {
                counter += 1;        // captured by value copy unless marked ref
        };

        ret add(2, 3);
}

## 16. Generics

// Monomorphized at compile time — every instantiation is a normal
// function/type, so there is zero runtime cost and cexpr folding
// works per instantiation.

function ::max<T>(a: T, b: T) -> T {
        ret if a > b { a } else { b };
}

struct ::vec<T> {
        pub {
                data: *T;
                len: uint32;
                cap: uint32;
        }
        #constructor self() {
                self.data = std::mem::dangling();   // raw pointer placeholder: valid
                                                    // only after assignment, and only
                                                    // dereferenceable inside volatile
                self.len = 0;
                self.cap = 0;
        }
}

function ::usage() {
        int32 m = ::max<int32>(3, 7);      // explicit args, inferred if obvious
        vec<float32> v = vec<float32> { }; // struct literal with generic args
        ret m;
}

// else is module-private. `extern` declares a foreign (C) symbol.

extern function ::uart_write(buf: []uint8, len: uint32) -> int32;   // links to C

pub function ::boot() { ... }        // public API of this module

## 18. SIMD and wide types

// Integer:   int8 int16 int32 int64   uint8 uint16 uint32 uint64
//            int128 int256 int512     uint128 uint256 uint512
// Float:     float16 float32 float64
//            float128 float256 float512
//
// There are no short aliases: int32 / uint32 / float64 are the only
// spellings (the u8"/u16" STRING prefixes are unaffected). Literal
// suffixes use the same names: 100int128, 1.5float64.
//
// Wide scalars lower to SIMD registers when the ISA has a wide enough
// one (AVX-512 for int512 lanes, SVE, RVV); otherwise to multi-limb
// software arithmetic on the target's largest registers, same code,
// chosen at compile time.

function ::wide() {
        int128 big  = 1000000000000000000000int128;
        int512 bigger = big * 1024;        // ZMM register or soft-limb fallback
        float128 f  = 1.5float128;
        ret;
}

// Vectorization control on loops (the compiler may still decline if
// illegal — it reports why instead of silently scalarizing):
//   #unroll(8)      force 8x unrolling (simple loops)
//   #vectorize      force vectorization (complex loops -> masks)
//   #novectorize    never touch this loop
#unroll(8)
for i in 0 .. 64 { crc = crc_table[crc ^ data[i]]; }

#vectorize
for i in 0 .. n { out[i] = a[i] * b[i] + c[i]; }

## 19. Constructible Matching

// Patterns are string literals with the p"..." prefix. p strings are
// raw (backslashes belong to the pattern). The compiler builds a
// linear-time (RE2-style) automaton at compile time — no backtracking,
// no allocation, worst-case O(n) — and lowers the match to target code
// (branch tree or SIMD instructions).

function ::is_ip_addr(ip_str: u8string) -> bool {
        ret match ip_str {
                // The compiler constructs the pattern as machine code,
                // generating optimized SIMD instructions where useful
                p"^(?:[0-9]{1,3}\.){3}[0-9]{1,3}$" {
                        true
                }
                ... {
                        false
                }
        }
}

// Patterns are compile-time constants and can be bound to variables:
let my_pattern = p"gr[ae]y";   // Maps to macro: std::<Pattern>

## 21. Packed structs

packed struct ::my_packed_struct {
        int32 a;
        bool b;
        int32 c;
        bool d;
        // the compiler will reorder to int32 int32 bool bool + 2 padding bytes
        // instead of int32 bool int32 bool (which wastes space due to padding)
        // NOTE: do not access fields via byte-level pointer arithmetic on packed
        // structs — that may cause undefined behavior inside volatile blocks.
        // Packed structs may improve performance: the compiler can keep the
        // entire struct data inside CPU registers (if it fits), and the CPU
        // may store it in the L1 cache.
}

## 24. Lexical structure (token reference for the lexer)

// Identifiers: [A-Za-z_][A-Za-z0-9_]*   (ASCII; unicode ids = future work)
// Line comment  // ...
// Block comment /* ... */  (nestable)
// Doc comment   /// ...   (retained only when #reflect is present)
//
// Reserved keywords:
//   as break cexpr continue defer else embed enum extern false fn for
//   function if in is let loop macro match module pub ref ret self
//   static struct true use volatile while packed interrupt
// Contextual (plain identifiers, meaningful only in position):
//   default  (only after #destructor)      main  (only in #addr(main))
//
// Punctuation / operators (maximal munch, longest first):
//   ::  ->  ..=  ..  &&  ||  ==  !=  <=  >=  <<  >>  <<=  >>=
//   +=  -=  *=  /=  %=  &=  |=  ^=  ??  ?  !  ~  +  -  *  /  %
//   ^  &  |  <  >  =  (  )  {  }  [  ]  ;  ,  .  :  #  $  |> ...
//
// Notes for the lexer:
//   - '::' then '<' forms a macro path (std::<println>); no combined token.
//   - '..' vs '.': scan '..' first (ranges), else '.' (member access).
//   - '$' and '^' are ordinary tokens; they are only special inside macro
//     patterns ($ = pattern start, ^ = pattern end) — parser decides.
//   - '<' / '>' serve both generics and comparison — parser resolves.
//   - Number suffixes (part of the literal token; the suffix IS the type
//     name): int8..int64 uint8..uint64 int128 uint128 int256 uint256
